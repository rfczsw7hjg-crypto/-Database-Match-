const express = require('express'); const bcrypt = require('bcryptjs'); const jwt = require('jsonwebtoken'); const sqlite3 = require('sqlite3').verbose(); const app = express(); const db = new sqlite3.Database('./database.sqlite'); const SECRET_KEY = 'fatima_secret_key_2026'; app.use(express.json()); // 1. إنشاء الجداول db.serialize(() => { db.run(`CREATE TABLE IF NOT EXISTS Users ( id INTEGER PRIMARY KEY AUTOINCREMENT, name TEXT NOT NULL, contact TEXT UNIQUE NOT NULL, password TEXT NOT NULL, role TEXT DEFAULT 'customer' )`); db.run(`CREATE TABLE IF NOT EXISTS Addresses ( id INTEGER PRIMARY KEY AUTOINCREMENT, user_id INTEGER NOT NULL, city TEXT NOT NULL, district TEXT NOT NULL, street TEXT NOT NULL, location_url TEXT, FOREIGN KEY (user_id) REFERENCES Users(id) ON DELETE CASCADE )`); }); // 2. التحقق من المدخلات والصلاحيات function validateInput(contact, password) { const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/; const phoneRegex = /^(05|5)(5|0|3|6|4|9|1|8|7)([0-9]{7})$/; if (!emailRegex.test(contact) && !phoneRegex.test(contact)) { return 'أدخل بريداً إلكترونياً صحيحاً أو رقم جوال سعودي (05xxxxxxxx)'; } if (!password || password.length < 6) { return 'كلمة المرور يجب أن تكون 6 خانات فأكثر'; } return null; } function authenticateToken(req, res, next) { const authHeader = req.headers['authorization']; req.user = user; next(); }); } function isAdmin(req, res, next) { if (req.user.role !== 'admin') { return res.status(403).json({ error: 'ممنوع: هذه الصلاحية للمشرف (Admin) فقط' }); } next(); } // 3. مسارات الـ API app.post('/api/register', async (req, res) => { const { name, contact, password, role } = req.body;
